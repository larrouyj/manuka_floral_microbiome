## 0.3 Helper functions: data import and cleaning ----------------------------

# Claident OTU summary: samples in rows, OTUs in columns
read_otu <- function(file) {
  x <- read.table(file, header = TRUE, sep = "\t", row.names = 1,
                  check.names = FALSE, quote = "", comment.char = "")
  otu_table(t(as.matrix(x)), taxa_are_rows = TRUE)
}

# Claident taxonomy; OTUs with these superkingdom values are dropped
read_tax <- function(file, drop_superkingdom = "") {
  x <- read.table(file, header = TRUE, sep = "\t", row.names = 1,
                  check.names = FALSE, quote = "", comment.char = "",
                  colClasses = "character")
  x <- x[!(x$superkingdom %in% drop_superkingdom), , drop = FALSE]
  tax_table(as.matrix(x))
}
drop_16S <- c("", "Chloroplast", "Mitochondria", "Uncertain")

read_meta <- function(file) {
  x <- read.csv(file, header = TRUE, fileEncoding = "UTF-8-BOM")
  names(x) <- sub("^(X\\.U\\.FEFF\\.|\u00ef\\.\\.)", "", names(x))  # stray BOM
  x
}

# Add plate position + run set. The first column of the plate file is the
# plate position (the original scripts renamed it "ple_ID" via gsub).
merge_plate <- function(meta, plate_file, breaks) {
  plate <- read_meta(plate_file)
  names(plate)[1] <- "Plate_position"
  m <- merge(meta, plate, by = "Sample_ID")
  m$runset <- cut(as.numeric(m$Plate_position), breaks = breaks,
                  labels = c("set1", "set2"))
  m
}

# Build phyloseq object; metadata matched to samples by Sample_ID when possible
attach_meta <- function(otu, tax, meta, id_col = "Sample_ID") {
  ids <- sample_names(otu)
  if (id_col %in% names(meta) && all(ids %in% meta[[id_col]])) {
    idx <- match(ids, meta[[id_col]])
    if (!identical(idx, seq_along(ids)))
      message("Metadata rows re-ordered to match the OTU table (matched by ID).")
    meta <- meta[idx, , drop = FALSE]
  } else {
    warning("Sample IDs not found in metadata: rows assigned by position, ",
            "as in the original scripts. Check the order carefully!")
    stopifnot(nrow(meta) == length(ids))
  }
  rownames(meta) <- ids
  phyloseq(otu, tax, sample_data(meta))   # keeps only OTUs present in 'tax'
}

# Samples x taxa count matrix
otu_mat <- function(ps) {
  m <- as(otu_table(ps), "matrix")
  if (taxa_are_rows(ps)) t(m) else m
}

keep_samples <- function(ps, var, levels) {
  prune_samples(as.character(sample_data(ps)[[var]]) %in% levels, ps)
}

# decontam, prevalence method, within one run set / year
find_contaminants <- function(ps, subset_var, subset_level, neg_type,
                              threshold = 0.5) {
  ps <- prune_samples(sample_sums(ps) >= 1, ps)
  ps <- keep_samples(ps, subset_var, subset_level)
  sample_data(ps)$is.neg <- sample_data(ps)$Sample_type == neg_type
  res <- isContaminant(ps, method = "prevalence", neg = "is.neg",
                       threshold = threshold)
  hits <- rownames(res)[which(res$contaminant)]
  cat(subset_level, "/", neg_type, ":", length(hits), "candidate OTU(s)\n")
  hits
}

# Remove contaminants, keep study samples, filter low-read samples and OTUs,
# remove OTUs unassigned at phylum level
clean_ps <- function(ps, bad_taxa = character(0), keep_var, keep_levels,
                     min_sample_reads = 0, min_taxon_reads = 2) {
  ps <- prune_taxa(!(taxa_names(ps) %in% bad_taxa), ps)
  ps <- keep_samples(ps, keep_var, keep_levels)
  ps <- prune_samples(sample_sums(ps) >= min_sample_reads, ps)
  ps <- prune_taxa(taxa_sums(ps) >= min_taxon_reads, ps)
  tt <- as.data.frame(as(tax_table(ps), "matrix"))
  ps <- prune_taxa(!is.na(tt$phylum) & !(tt$phylum %in% c("", "uncharacterized")), ps)
  print(ps); cat("Total reads:", sum(sample_sums(ps)), "\n")
  ps
}

# Write processed OTU table, taxonomy and metadata (files for deposit)
export_ps <- function(ps, prefix) {
  w <- function(x, what)
    write.csv(x, file.path(out_dir, paste0(prefix, "_", what, ".csv")))
  w(as(otu_table(ps), "matrix"), "OTU_table")
  w(as(tax_table(ps), "matrix"), "taxonomy")
  w(as(sample_data(ps), "data.frame"), "metadata")
}

# Fill unassigned ranks with the lowest assigned rank (for plot labels only)
fill_tax <- function(ps) {
  ranks  <- c("superkingdom", "phylum", "class", "order", "family", "genus", "species")
  prefix <- c("Kingdom_", "Phylum_", "Class_", "Order_", "Family_", "Genus_")
  tt <- as.data.frame(as(tax_table(ps), "matrix"), stringsAsFactors = FALSE)[, ranks]
  tt[is.na(tt)] <- ""
  for (i in seq_len(nrow(tt))) for (r in 2:7) {
    if (tt[i, r] == "") { tt[i, r:7] <- paste0(prefix[r - 1], tt[i, r - 1]); break }
  }
  tax_table(ps) <- tax_table(as.matrix(tt))
  ps
}

# Move the legend into the empty facet panel (was shift_legend2())
legend_in_empty_panel <- function(p) {
  gp <- ggplotGrob(p)
  panels <- grep("^panel", gp$layout$name)
  empty  <- panels[sapply(panels, function(i) inherits(gp$grobs[[i]], "zeroGrob"))]
  lemon::reposition_legend(p, "center", panel = gp$layout$name[empty], plot = FALSE)
}

# Core OTUs: detection = 0.001, prevalence = 75 %
core_by_group <- function(ps, group_var, levels, prefix,
                          detection = 0.001, prevalence = 0.75) {
  out <- do.call(rbind, lapply(levels, function(l) {
    rel <- microbiome::transform(keep_samples(ps, group_var, l), "compositional")
    ids <- core_members(rel, detection = detection, prevalence = prevalence)
    if (length(ids) == 0) return(NULL)
    data.frame(group = l, OTU = ids,
               as.data.frame(as(tax_table(rel), "matrix"))[ids, , drop = FALSE],
               row.names = NULL, check.names = FALSE)
  }))
  write.csv(out, file.path(out_dir, paste0(prefix, "_core_OTUs.csv")), row.names = FALSE)
  print(table(out$group))
  out
}

# Shared/unique OTUs between groups (UpSet plot)
plot_upset <- function(ps, group_var, sets, labels = sets, set_cols,
                       ylab = "Unique or shared OTUs", xlab = "OTUs per Niche",
                       text_scale = 1.9, ...) {
  m   <- merge_samples(ps, group_var)
  otu <- as(otu_table(m), "matrix"); if (!taxa_are_rows(m)) otu <- t(otu)
  bin <- as.data.frame((otu[, sets, drop = FALSE] > 0) * 1L)
  colnames(bin) <- labels
  upset(bin, sets = rev(labels), keep.order = TRUE, nsets = length(labels),
        order.by = "freq", text.scale = text_scale,
        mainbar.y.label = ylab, sets.x.label = xlab,
        sets.bar.color = rev(set_cols), ...)
}

# Shannon index (vegan) + metadata
add_shannon <- function(ps_t) {
  df <- as(sample_data(ps_t), "data.frame")
  df$Shannon <- vegan::diversity(otu_mat(ps_t), index = "shannon")
  df
}

# LMM: Shannon ~ niche + (1 | plant); LRT against null model; Tukey post hoc
shannon_lmm <- function(ps_t, levels, name, group_var = "Sample_type",
                        random_var = "Plant") {
  df <- add_shannon(ps_t)
  df$group  <- factor(as.character(df[[group_var]]), levels = levels)
  df$random <- factor(df[[random_var]])
  full <- lmer(Shannon ~ group + (1 | random), data = df)
  null <- lmer(Shannon ~ 1 + (1 | random), data = df)
  lrt  <- anova(null, full)                     # ML refit, chi-square LRT
  ph   <- as.data.frame(pairs(emmeans(full, ~ group)))
  save_txt(list(LRT = lrt, Pairwise = ph), name)
  print(lrt)
  # Model checks: plot(full); qqnorm(resid(full)); qqline(resid(full))
  list(data = df, model = full, lrt = lrt, posthoc = ph)
}

plot_shannon <- function(df, levels, labels, cols) {
  df$group <- factor(as.character(df$group), levels = levels, labels = labels)
  set.seed(seed)
  ggplot(df, aes(group, Shannon, colour = group)) +
    geom_boxplot(outlier.shape = NA) + geom_jitter(alpha = 0.6) +
    scale_colour_manual(values = cols) +
    labs(x = "Niches", y = "Shannon index") + theme_fig +
    theme(axis.text.x = element_blank(), axis.ticks.x = element_blank())
}

# Bray-Curtis PERMANOVA (permutations within plant), dispersion, pairwise tests
beta_tests <- function(ps_t, group_var, name, strata_var = "Plant",
                       pairwise = FALSE) {
  meta <- as(sample_data(ps_t), "data.frame")
  d <- phyloseq::distance(ps_t, method = "bray")
  strata <- if (is.null(strata_var)) NULL else meta[[strata_var]]
  set.seed(seed)
  perm <- adonis2(as.formula(paste("d ~", group_var)), data = meta,
                  permutations = 999, strata = strata)
  set.seed(seed)
  disp <- permutest(betadisper(d, meta[[group_var]]), permutations = 999)
  pw <- if (pairwise) pairwise.adonis(d, meta[[group_var]]) else NULL
  save_txt(list(PERMANOVA = perm, Dispersion = disp, Pairwise = pw), name)
  print(perm); print(disp)
  invisible(list(permanova = perm, dispersion = disp, pairwise = pw))
}

# NMDS (Bray-Curtis); metaMDS() defaults kept as in the original analysis
plot_nmds <- function(ps_t, group_var, levels, labels = levels, cols,
                      mean_point = FALSE, star_plot = FALSE) {
  set.seed(seed)
  ord <- metaMDS(otu_mat(ps_t), distance = "bray", trymax = 30000, trace = FALSE)
  cat("NMDS stress:", round(ord$stress, 4), "\n")
  sc  <- as.data.frame(scores(ord, display = "sites"))
  sc$group <- factor(as.character(sample_data(ps_t)[[group_var]]),
                     levels = levels, labels = labels)
  ggscatter(sc, x = "NMDS1", y = "NMDS2", color = "group", ellipse = TRUE,
            ellipse.border.remove = TRUE, mean.point = mean_point,
            mean.point.size = 6, star.plot = star_plot) +
    scale_color_manual(values = cols) + scale_fill_manual(values = cols) +
    theme_fig
}

# Multivariate GLM (negative binomial) on raw counts, PIT-trap resampling
# (slow: ~1.5 h for the 16S niche data set)
run_manyglm <- function(ps, group_var, name, nBoot = 999) {
  df <- as(sample_data(ps), "data.frame")
  df$group <- factor(df[[group_var]])
  y <- mvabund(otu_mat(ps))
  mod <- manyglm(y ~ group, data = df, family = "negative_binomial")
  set.seed(seed)
  res <- anova(mod, nBoot = nBoot)
  save_txt(res, name); print(res)
  invisible(list(model = mod, anova = res))
}

# Top-n OTUs per group, pooled by group, remainder = "Others" (Fig. 1C/D)
plot_top_merged <- function(ps, group_var, n = 20, rank = "genus") {
  rel <- transform_sample_counts(merge_samples(fill_tax(ps), group_var),
                                 function(x) 100 * x / sum(x))
  top <- names(sort(taxa_sums(rel), decreasing = TRUE))[seq_len(n)]
  df  <- psmelt(prune_taxa(top, rel))
  df$taxon <- df[[rank]]
  df  <- aggregate(Abundance ~ Sample + taxon, data = df, FUN = sum)
  oth <- aggregate(Abundance ~ Sample, data = df, FUN = sum)
  oth$Abundance <- 100 - oth$Abundance; oth$taxon <- "Others"
  df  <- rbind(df, oth[, names(df)])
  df$taxon <- factor(df$taxon,
                     levels = c(sort(setdiff(unique(df$taxon), "Others")), "Others"))
  set.seed(seed)
  pal <- c(distinctColorPalette(nlevels(df$taxon) - 1), "grey80")
  ggplot(df, aes(Sample, Abundance, fill = taxon)) + geom_col() +
    scale_fill_manual(values = pal) +
    labs(x = NULL, y = "Relative abundance (%)", fill = rank) + theme_classic() +
    theme(legend.text  = element_text(size = 11, face = "bold.italic"),
          legend.title = element_text(size = 15, face = "bold"),
          axis.title.y = element_text(size = 13, face = "bold"),
          axis.text.x  = element_text(angle = 45, hjust = 1))
}

# Top-n OTUs within each niche, one bar per sample (Fig. 3, 4)
plot_top_by_niche <- function(ps, levels, labels, n = 20, rank = "genus") {
  ps <- fill_tax(ps)   # unresolved genera labelled by lowest known rank
  df <- do.call(rbind, lapply(levels, function(g) {
    rel <- transform_sample_counts(keep_samples(ps, "Sample_type", g),
                                   function(x) 100 * x / sum(x))
    top <- names(sort(taxa_sums(rel), decreasing = TRUE))[seq_len(n)]
    psmelt(tax_glom(prune_taxa(top, rel), taxrank = rank, NArm = FALSE))
  }))
  df$Sample_type <- factor(df$Sample_type, levels = levels, labels = labels)
  set.seed(seed)
  pal <- distinctColorPalette(length(unique(df[[rank]])))
  ggplot(df, aes(Sample, Abundance, fill = .data[[rank]])) +
    geom_col() +
    facet_wrap(~ Sample_type, ncol = 3, scales = "free_x") +
    scale_fill_manual(values = pal) +
    labs(x = NULL, y = "Relative abundance (%)", fill = rank) +
    theme_classic() +
    theme(axis.text.x = element_blank(), axis.ticks.x = element_blank(),
          legend.direction = "horizontal",
          legend.text  = element_text(size = 11, face = "bold.italic"),
          legend.title = element_text(size = 15, face = "bold"),
          strip.text   = element_text(size = 11, face = "bold"),
          axis.title.y = element_text(size = 13, face = "bold"))
}
